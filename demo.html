// Design token resolver
const resolveToken = (path, theme = 'light') => {
  const tokens = loadTokens(theme);
  return path.split('.').reduce(
    (obj, key) => obj?.[key], tokens
  );
};

const teal   = resolveToken('color.teal');      // #06B6D4
const radius = resolveToken('border.radius.lg');  // 16px
