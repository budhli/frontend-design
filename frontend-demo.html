// Design token resolver
const resolveToken = (path, theme = 'light') => {
  const tokens = loadTokens(theme);
  return path.split('.').reduce(
    (obj, key) => obj?.[key],
    tokens
  );
};

// Usage
const accent = resolveToken('color.accent'); // #C8922A
const radius = resolveToken('border.radius.lg'); // 12px
